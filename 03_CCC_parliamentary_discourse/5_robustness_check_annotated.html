Input and corpus version

The main climate-policy corpus is constructed using word-boundary keyword matching. This file compares the original CCC policy-theme dictionary with a stricter dictionary specification. The stricter dictionary removes selected keywords with lower contextual validity and is used only for robustness checks.

Robustness specification

This file uses a stricter matching strategy and/or stricter dictionary specification. These results are used as robustness checks and do not replace the main analysis.

Workflow note

This document checks whether the period-level and yearly salience patterns depend on the specification of the CCC policy-theme dictionary. It reconstructs the corpus, applies the original and stricter keyword sets, and compares their matched proportions over time. Differences between specifications indicate sensitivity to keyword coverage and matching decisions; they are not independent estimates of a causal CCC effect. All original code is preserved.

# Carry out the 'helper function' step and inspect its output before proceeding. This chunk forms part of the documented workflow; its results have not been independently re-run here.
# Create one combined word-boundary regex pattern from a keyword vector

make_pattern <- function(keywords) {
  
  pattern <- paste(
    keywords,
    collapse = "|"
  )
  
  pattern <- paste0(
    "\\b(",
    pattern,
    ")\\b"
  )
  
  return(pattern)
}

# Load the input data and construct the objects required by subsequent processing steps.

all_files <- list.files("xml_folder_path",
                    pattern = "\\.xml$",
                    full.names = TRUE)

parse_one_xml <- function(file_path) {
 doc <- read_xml(file_path)
 xml_ns_strip(doc)
 
 if (length(xml_find_all(doc, ".//paragraphe")) > 0) {
  nodes <- xml_find_all(doc, ".//paragraphe")
  tag_used <- "paragraphe"
 } else if (length(xml_find_all(doc, ".//para")) > 0) {
  nodes <- xml_find_all(doc, ".//para")
  tag_used <- "para"
 } else if (length(xml_find_all(doc, ".//texte")) > 0) {
  nodes <- xml_find_all(doc, ".//texte")
  tag_used <- "texte"
 } else {
  return(tibble(
         file = basename(file_path),
         tag_used = NA_character_,
         speech_text = NA_character_
         ))
 }
 
tibble(
 file = basename(file_path),
 tag_used = tag_used,
 speech_text = xml_text(nodes)
) %>%
 mutate(
  speech_text = str_remove_all(speech_text, "\\d+"),
  speech_text = str_remove_all(speech_text, "\\([^\\)]+\\)"),
  speech_text = str_remove_all(speech_text, "La parole est à.*?(\\.|$)"),
  speech_text = str_remove_all(speech_text, "La séance est.*?(\\.|$)"),
  speech_text = str_squish(speech_text)
 ) %>%
 filter(!is.na(speech_text), speech_text != "")
}

# Apply the specified terms to speech texts and create the corresponding match indicator or subset. Verify the unit of analysis before interpreting the resulting counts.

original_pattern <- make_pattern(original_dict)
strict_pattern <- make_pattern(strict_dict)

df_policy_versions <- df_climate %>%
 mutate(
  text_lower = str_to_lower(speech_text),
  
  original_match = str_detect(
   text_lower,
   regex(original_pattern, ignore_case = TRUE)
  ),
  
  strict_match = str_detect(
   text_lower,
   regex(strict_pattern, ignore_case = TRUE)
  )
 )

period-level ratio: original vs stricter

# Calculate or display the relevant period-level proportions. The interpretation depends on the numerator, denominator and dictionary specification used in this chunk.

period_ratio <- df_policy_versions %>%
 group_by(period) %>%
 summarise(
  climate_total = n(),
  
  original_raw_count = sum(original_match, na.rm = TRUE),
  strict_raw_count = sum(strict_match, na.rm = TRUE),
  .groups = "drop"
 ) %>%
 mutate(
  original_proportion = original_raw_count / climate_total,
  strict_proportion = strict_raw_count / climate_total,
  original_percentage = round(original_proportion * 100, 2),
  strict_percentage = round(strict_proportion * 100, 2),
  original_proportion = round(original_proportion, 3),
  strict_proportion = round(strict_proportion, 3),
  period = factor(period, levels = c("pre", "transition", "post")),
 ) %>%
 arrange(period)
 
write.csv(period_ratio, "period_ratio_original_vs_strict.csv", row.names = FALSE)
